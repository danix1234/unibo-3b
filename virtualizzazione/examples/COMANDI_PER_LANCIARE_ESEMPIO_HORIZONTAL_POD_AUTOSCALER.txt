
APRO UN TERMINALE E VADO NELLA DIRECTORY DI MINUKUBE

cd /cygdrive/c/DIDATTICA/SystemsIntegration_AmministrazioneDiSistemi/DISPENSE/MINIKUBE

INIZIALIZZO IL CLUSTER MINIKUBE

minikube stop
minikube start --extra-config=kubelet.housekeeping-interval=10s
minikube addons list
minikube addons enable metrics-server

FACCIO PARTIRE L'APPLICAZIONE BASE

kubectl apply -f https://k8s.io/examples/application/php-apache.yaml
kubectl get pods
kubectl get pods --all-namespaces
kubectl get hpa --all-namespaces

CREO l'AUTOSCALER ORRIZZONTALE

kubectl autoscale deployment php-apache --cpu-percent=50 --min=1 --max=10
kubectl get hpa --all-namespaces
kubectl get hpa php-apache --watch


IN UN ALTRO TERMINALE LANCIO IL GENERATORE DI CARICO
kubectl run -i --tty load-generator --rm --image=busybox:1.28 --restart=Never -- /bin/sh -c "while sleep 0.01; do wget -q -O- http://php-apache; done"

kubectl get hpa php-apache --watch
